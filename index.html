function doGet() { const data = getDashboardData(); return ContentService .createTextOutput(JSON.stringify({ status: "success", data: data })) .setMimeType(ContentService.MimeType.JSON); } // ========================================================== // DASHBOARD DATA // ========================================================== function getDashboardData() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName("Dashboard Data"); const data = sheet.getDataRange().getValues(); const headers = data[0]; const rows = data .slice(1) .filter(row => row[1] !== ""); return [ headers, ...rows ]; } // ========================================================== // SAVE DAILY FARM ENTRY // ========================================================== function doPost(e) { try { const data = JSON.parse(e.postData.contents); const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName("Daily Farm Entry"); if (!sheet) { throw new Error( "Daily Farm Entry sheet not found" ); } // ------------------------------------------------------ // INPUTS // ------------------------------------------------------ const date = data.date || ""; const batchId = data.batchId || ""; const farmName = data.farmName || ""; const age = data.age || ""; const birds = data.birds || ""; const mortality = data.mortality || 0; const feedType = data.feedType || ""; const remainingFeedStock = data.remainingFeedStock || ""; // ------------------------------------------------------ // VALIDATION // ------------------------------------------------------ if (!date) { throw new Error( "Date is required" ); } if (!batchId) { throw new Error( "Batch ID is required" ); } if (!farmName) { throw new Error( "Farm name is required" ); } // ------------------------------------------------------ // SAVE ROW // ------------------------------------------------------ sheet.appendRow([ date, batchId, farmName, age, birds, mortality, feedType, remainingFeedStock ]); // ------------------------------------------------------ // RESPONSE // ------------------------------------------------------ return ContentService .createTextOutput(JSON.stringify({ status: "success", message: "Daily entry saved successfully" })) .setMimeType( ContentService.MimeType.JSON ); } catch (error) { return ContentService .createTextOutput(JSON.stringify({ status: "error", message: error.message })) .setMimeType( ContentService.MimeType.JSON ); } }
